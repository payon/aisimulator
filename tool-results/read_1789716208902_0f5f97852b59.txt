   165→// === 툴팁 위치 계산 ===
   166→interface TooltipPosition {
   167→  top: number;
   168→  left: number;
   169→  arrowPosition: 'top' | 'bottom' | 'left' | 'right';
   170→}
   171→
   172→function calculateTooltipPosition(
   173→  targetRect: DOMRect,
   174→  position: GuideStep['position'],
   175→  tooltipWidth: number,
   176→  tooltipHeight: number
   177→): TooltipPosition {
   178→  const gap = 16;
   179→  const vw = window.innerWidth;
   180→  const vh = window.innerHeight;
   181→  const padding = 12;
   182→
   183→  let top = 0;
   184→  let left = 0;
   185→  let arrowPosition = position;
   186→
   187→  switch (position) {
   188→    case 'bottom':
   189→      top = targetRect.bottom + gap;
   190→      left = targetRect.left + targetRect.width / 2 - tooltipWidth / 2;
   191→      break;
   192→    case 'top':
   193→      top = targetRect.top - tooltipHeight - gap;
   194→      left = targetRect.left + targetRect.width / 2 - tooltipWidth / 2;
   195→      break;
   196→    case 'right':
   197→      top = targetRect.top + targetRect.height / 2 - tooltipHeight / 2;
   198→      left = targetRect.right + gap;
   199→      break;
   200→    case 'left':
   201→      top = targetRect.top + targetRect.height / 2 - tooltipHeight / 2;
   202→      left = targetRect.left - tooltipWidth - gap;
   203→      break;
   204→  }
   205→
   206→  // Clamp to viewport
   207→  left = Math.max(padding, Math.min(left, vw - tooltipWidth - padding));
   208→  top = Math.max(padding, Math.min(top, vh - tooltipHeight - padding));
   209→
   210→  // Fallback: if tooltip would be off-screen, reposition
   211→  if (position === 'bottom' && top + tooltipHeight > vh - padding) {
   212→    top = targetRect.top - tooltipHeight - gap;
   213→    arrowPosition = 'top';
   214→  }
   215→  if (position === 'top' && top < padding) {
   216→    top = targetRect.bottom + gap;
   217→    arrowPosition = 'bottom';
   218→  }
   219→  if (position === 'right' && left + tooltipWidth > vw - padding) {
   220→    left = targetRect.left - tooltipWidth - gap;
   221→    arrowPosition = 'left';
   222→  }
   223→  if (position === 'left' && left < padding) {
   224→    left = targetRect.right + tooltipWidth + gap;
   225→    arrowPosition = 'right';
   226→  }
   227→
   228→  return { top, left, arrowPosition };
   229→}

   231→// === 화살표 아이콘 선택 ===
   232→function ArrowIcon({ direction, className }: { direction: GuideStep['position']; className?: string }) {
   233→  switch (direction) {
   234→    case 'top':
   235→      return <ChevronUp className={className} />;
   236→    case 'bottom':
   237→      return <ChevronDown className={className} />;
   238→    case 'left':
   239→      return <ChevronLeft className={className} />;
   240→    case 'right':
   241→      return <ChevronRight className={className} />;
   242→  }
   243→}

   245→// === 메인 컴포넌트 ===
   246→export default function SeniorGuideTour() {
   247→  const { active, currentStep, completed, start, stop, nextStep, prevStep, complete } = useGuideTourStore();
   248→  const { voiceEnabled, readingSpeed } = useSettingsStore();
   249→  const { speak, stop: stopTTS } = useSpeechSynthesis();
   250→
   251→  // 사이드바 제어 (모바일에서 자동 열기/닫기)
   252→  const { isMobile, openMobile, setOpenMobile } = useSidebar();
   253→
   254→  const [targetRect, setTargetRect] = useState<DOMRect | null>(null);
   255→  const [tooltipPos, setTooltipPos] = useState<TooltipPosition>({ top: 0, left: 0, arrowPosition: 'bottom' });
   256→  const [showWelcome, setShowWelcome] = useState(false);
   257→  const [showComplete, setShowComplete] = useState(false);
   258→  const [stepNotFound, setStepNotFound] = useState(false);
   259→  const [tooltipSize, setTooltipSize] = useState({ width: 320, height: 200 });
   260→
   261→  // 투어 중 사이드바를 자동으로 열었는지 추적 → 완료/스킵 시 원래대로 복원
   262→  const sidebarAutoOpenedRef = useRef(false);
   263→
   264→  const tooltipRef = useRef<HTMLDivElement>(null);
   265→  const resizeObserverRef = useRef<ResizeObserver | null>(null);
   266→  const rafRef = useRef<number | null>(null);
   267→  const retryTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   268→  // 자기 자신을 재시도할 때 사용하는 ref (circular dependency 해결)
   269→  const syncGuidePositionRef = useRef<() => void>(() => {});
   270→
   271→  // 현재 디바이스에 맞는 스텝 배열
   272→  const steps = isMobile ? MOBILE_GUIDE_STEPS : DESKTOP_GUIDE_STEPS;
   273→
   274→  // Start tour -> show welcome screen
   275→  const handleStartTour = useCallback(() => {
   276→    setShowWelcome(true);
   277→  }, []);
   278→
   279→  // Welcome -> begin actual tour
   280→  const handleBeginTour = useCallback(() => {
   281→    setShowWelcome(false);
   282→    start();
   283→  }, [start]);
   284→
   285→  // Skip from welcome
   286→  const handleSkipWelcome = useCallback(() => {
   287→    setShowWelcome(false);
   288→    complete();
   289→  }, [complete]);
   290→
   291→  // 모바일 사이드바 자동 열기 (필요한 스텝에서 호출)
   292→  const ensureSidebarOpen = useCallback(() => {
   293→    if (isMobile && !openMobile) {
   294→      setOpenMobile(true);
   295→      sidebarAutoOpenedRef.current = true;
   296→    }
   297→  }, [isMobile, openMobile, setOpenMobile]);
   298→
   299→  // 모바일 사이드바 자동 닫기 (투어 종료 시)
   300→  const closeAutoOpenedSidebar = useCallback(() => {
   301→    if (sidebarAutoOpenedRef.current && isMobile && openMobile) {
   302→      setOpenMobile(false);
   303→      sidebarAutoOpenedRef.current = false;
   304→    }
   305→  }, [isMobile, openMobile, setOpenMobile]);
   306→
   307→  // Combined: find target, update rect, and calculate tooltip position
   308→  // Uses requestAnimationFrame to avoid synchronous setState in effects
   309→  const syncGuidePosition = useCallback(() => {
   310→    if (!active || showWelcome || showComplete) return;
   311→
   312→    const step = steps[currentStep];
   313→    if (!step) {
   314→      setShowComplete(true);
   315→      return;
   316→    }
   317→
   318→    // 모바일에서 사이드바가 필요한 스텝인데 사이드바가 닫혀 있으면 열고 재시도
   319→    if (step.requiresSidebarOpen && isMobile && !openMobile) {
   320→      ensureSidebarOpen();
   321→      // 사이드바 Sheet 애니메이션 완료 후 재시도 (350ms 대기)
   322→      if (retryTimerRef.current) clearTimeout(retryTimerRef.current);
   323→      retryTimerRef.current = setTimeout(() => {
   324→        syncGuidePositionRef.current();
   325→      }, 350);
   326→      return;
   327→    }
   328→
   329→    const el = document.querySelector(step.targetSelector) as HTMLElement | null;
   330→    if (!el) {
   331→      // 모바일에서 사이드바 내부 요소를 못 찾으면 사이드바를 열고 재시도
   332→      if (step.requiresSidebarOpen && isMobile) {
   333→        ensureSidebarOpen();
   334→        if (retryTimerRef.current) clearTimeout(retryTimerRef.current);
   335→        retryTimerRef.current = setTimeout(() => {
   336→          syncGuidePositionRef.current();
   337→        }, 350);
   338→        return;
   339→      }
   340→
   341→      setStepNotFound(true);
   342→      if (currentStep < steps.length - 1) {
   343→        nextStep();
   344→      } else {
   345→        setShowComplete(true);
   346→      }
   347→      return;
   348→    }
   349→
   350→    setStepNotFound(false);
   351→    const rect = el.getBoundingClientRect();
   352→    setTargetRect(rect);
   353→
   354→    // Scroll element into view if needed
   355→    const isInViewport =
   356→      rect.top >= 0 &&
   357→      rect.left >= 0 &&
   358→      rect.bottom <= window.innerHeight &&
   359→      rect.right <= window.innerWidth;
   360→
   361→    if (!isInViewport) {
   362→      el.scrollIntoView({ behavior: 'smooth', block: 'center', inline: 'center' });
   363→    }
   364→
   365→    // Calculate tooltip position after getting rect
   366→    // Use rAF for the scroll to settle
   367→    rafRef.current = requestAnimationFrame(() => {
   368→      const finalRect = el.getBoundingClientRect();
   369→      setTargetRect(finalRect);
   370→
   371→      const tw = tooltipSize.width;
   372→      const th = tooltipSize.height;
   373→      const pos = calculateTooltipPosition(finalRect, step.position, tw, th);
   374→      setTooltipPos(pos);
   375→    });
   376→  }, [active, currentStep, showWelcome, showComplete, nextStep, tooltipSize, steps, isMobile, openMobile, ensureSidebarOpen]);
   377→
   378→  // ref 업데이트 (setTimeout 재시도에서 사용)
   379→  useEffect(() => {
   380→    syncGuidePositionRef.current = syncGuidePosition;
   381→  }, [syncGuidePosition]);
   382→
   383→  // Measure tooltip size when step changes
   384→  useEffect(() => {
   385→    if (tooltipRef.current) {
   386→      const { width, height } = tooltipRef.current.getBoundingClientRect();
   387→      setTooltipSize({ width, height });
   388→    }
   389→  }, [active, currentStep, showWelcome, showComplete]);
   390→
   391→  // Sync position on step change using rAF to avoid lint warning
   392→  useEffect(() => {
   393→    rafRef.current = requestAnimationFrame(() => {
   394→      syncGuidePosition();
   395→    });
   396→    return () => {
   397→      if (rafRef.current !== null) {
   398→        cancelAnimationFrame(rafRef.current);
   399→      }
   400→      if (retryTimerRef.current !== null) {
   401→        clearTimeout(retryTimerRef.current);
   402→      }
   403→    };
   404→  }, [syncGuidePosition]);
   405→
   406→  // Resize observer + window resize/scroll listener for target element
   407→  useEffect(() => {
   408→    if (!active || showWelcome || showComplete) return;
   409→
   410→    const step = steps[currentStep];
   411→    if (!step) return;
   412→
   413→    const el = document.querySelector(step.targetSelector) as HTMLElement | null;
   414→    if (!el) return;
   415→
   416→    resizeObserverRef.current = new ResizeObserver(() => {
   417→      const rect = el.getBoundingClientRect();
   418→      setTargetRect(rect);
   419→      const tw = tooltipSize.width;
   420→      const th = tooltipSize.height;
   421→      const pos = calculateTooltipPosition(rect, step.position, tw, th);
   422→      setTooltipPos(pos);
   423→    });
   424→    resizeObserverRef.current.observe(el);
   425→
   426→    const handleResize = () => {
   427→      requestAnimationFrame(() => syncGuidePosition());
   428→    };
   429→    const handleScroll = () => {
   430→      const rect = el.getBoundingClientRect();
   431→      setTargetRect(rect);
   432→      const tw = tooltipSize.width;
   433→      const th = tooltipSize.height;
   434→      const pos = calculateTooltipPosition(rect, step.position, tw, th);
   435→      setTooltipPos(pos);
   436→    };
   437→
   438→    window.addEventListener('resize', handleResize);
   439→    window.addEventListener('scroll', handleScroll, true);
   440→
   441→    return () => {
   442→      resizeObserverRef.current?.disconnect();
   443→      window.removeEventListener('resize', handleResize);
   444→      window.removeEventListener('scroll', handleScroll, true);
   445→    };
   446→  }, [active, currentStep, showWelcome, showComplete, syncGuidePosition, tooltipSize, steps]);
   447→
   448→  // TTS: read step description aloud when step changes
   449→  useEffect(() => {
   450→    if (!active || showWelcome || showComplete) return;
   451→    if (!voiceEnabled) return;
   452→
   453→    const step = steps[currentStep];
   454→    if (!step) return;
   455→
   456→    const timer = setTimeout(() => {
   457→      speak(step.description, readingSpeed);
   458→    }, 600);
   459→
   460→    return () => {
   461→      clearTimeout(timer);
   462→      stopTTS();
   463→    };
   464→  }, [active, currentStep, voiceEnabled, showWelcome, showComplete, speak, stopTTS, readingSpeed, steps]);
   465→
   466→  // Handle next step
   467→  const handleNext = useCallback(() => {
   468→    stopTTS();
   469→    if (currentStep >= steps.length - 1) {
   470→      setShowComplete(true);
   471→    } else {
   472→      nextStep();
   473→    }
   474→  }, [currentStep, nextStep, stopTTS, steps]);
   475→
   476→  // Handle previous step
   477→  const handlePrev = useCallback(() => {
   478→    stopTTS();
   479→    prevStep();
   480→  }, [prevStep, stopTTS]);
   481→
   482→  // Handle skip
   483→  const handleSkip = useCallback(() => {
   484→    stopTTS();
   485→    stop();
   486→    complete();
   487→    closeAutoOpenedSidebar();
   488→  }, [stop, complete, stopTTS, closeAutoOpenedSidebar]);
   489→
   490→  // Handle complete
   491→  const handleComplete = useCallback(() => {
   492→    stopTTS();
   493→    setShowComplete(false);
   494→    complete();
   495→    closeAutoOpenedSidebar();
   496→  }, [complete, stopTTS, closeAutoOpenedSidebar]);
   497→
   498→  // Current step data
   499→  const step = steps[currentStep];
   500→  const totalSteps = steps.length;
   501→
   502→  return (
   503→    <>
   504→      {/* 도움말 버튼 (투어가 비활성일 때 표시) */}
   505→      {!active && !showWelcome && !showComplete && (
   506→        <motion.button
