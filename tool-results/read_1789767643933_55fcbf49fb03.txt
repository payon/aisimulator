   230→export default function MockDataManager() {
   231→  const { authenticatedFetch } = useAdminAuth();
   232→  const [activeTab, setActiveTab] = useState<MockTab>('chat');
   233→  const [mockData, setMockData] = useState<MockData>({

   239→  const [isLoading, setIsLoading] = useState(true);
   240→  const [isSaving, setIsSaving] = useState(false);
   241→  const [isResetting, setIsResetting] = useState(false);
   242→  const [hasChanges, setHasChanges] = useState(false);
   243→
   244→  // 목업 데이터 로드
   245→  const loadMockData = useCallback(async () => {
   246→    try {
   247→      setIsLoading(true);
   248→      const res = await authenticatedFetch('/api/admin/mock');
   249→      if (res.ok) {
   250→        const data = await res.json();
   251→        if (data.mockData) {
   252→          setMockData(data.mockData);
   253→          setHasChanges(false);
   254→        }
   255→      }
   256→    } catch {
   257→      toast.error('목업 데이터를 불러오는데 실패했습니다.');
   258→    } finally {
   259→      setIsLoading(false);
   260→    }
   261→  }, [authenticatedFetch]);
   262→
   263→  useEffect(() => {
   264→    loadMockData();
   265→  }, [loadMockData]);
   266→
   267→  // 저장
   268→  const handleSave = async () => {
   269→    setIsSaving(true);
   270→    try {
   271→      const res = await authenticatedFetch('/api/admin/mock', {
   272→        method: 'POST',
   273→        body: JSON.stringify({ mockData }),
   274→      });
   275→      if (res.ok) {
   276→        toast.success('목업 데이터가 저장되었습니다.');
   277→        setHasChanges(false);
   278→        // 프론트엔드에 즉시 반영
   279→        refreshCmsContent();
   280→      } else {
   281→        const data = await res.json();
   282→        toast.error(data.error || '저장에 실패했습니다.');
   283→      }
   284→    } catch {
   285→      toast.error('저장 중 오류가 발생했습니다.');
   286→    } finally {
   287→      setIsSaving(false);
   288→    }
   289→  };
   290→
   291→  // 기본값으로 초기화
   292→  const handleReset = async () => {
   293→    setIsResetting(true);
   294→    try {
   295→      const res = await authenticatedFetch('/api/admin/mock/reset', {
   296→        method: 'POST',
