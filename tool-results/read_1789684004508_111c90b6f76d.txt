   229→export default function MockDataManager() {
   230→  const { authenticatedFetch } = useAdminAuth();
   231→  const [activeTab, setActiveTab] = useState<MockTab>('chat');
   232→  const [mockData, setMockData] = useState<MockData>({
   233→    chat: DEFAULT_CHAT,
   234→    image: DEFAULT_IMAGE,
   235→    future: DEFAULT_FUTURE,
   236→    quiz: DEFAULT_QUIZ,
   237→  });
   238→  const [isLoading, setIsLoading] = useState(true);
   239→  const [isSaving, setIsSaving] = useState(false);
   240→  const [isResetting, setIsResetting] = useState(false);
   241→  const [hasChanges, setHasChanges] = useState(false);
   242→
   243→  // 목업 데이터 로드
   244→  const loadMockData = useCallback(async () => {
   245→    try {
   246→      setIsLoading(true);
   247→      const res = await authenticatedFetch('/api/admin/mock');
   248→      if (res.ok) {
   249→        const data = await res.json();
   250→        if (data.mockData) {
   251→          setMockData(data.mockData);
   252→          setHasChanges(false);
   253→        }
   254→      }
   255→    } catch {
   256→      toast.error('목업 데이터를 불러오는데 실패했습니다.');
   257→    } finally {
   258→      setIsLoading(false);
   259→    }
   260→  }, [authenticatedFetch]);
   261→
   262→  useEffect(() => {
   263→    loadMockData();
   264→  }, [loadMockData]);
   265→
   266→  // 저장
   267→  const handleSave = async () => {
   268→    setIsSaving(true);
   269→    try {
   270→      const res = await authenticatedFetch('/api/admin/mock', {
   271→        method: 'POST',
   272→        body: JSON.stringify({ mockData }),
   273→      });
   274→      if (res.ok) {
   275→        toast.success('목업 데이터가 저장되었습니다.');
   276→        setHasChanges(false);
   277→      } else {
   278→        const data = await res.json();
   279→        toast.error(data.error || '저장에 실패했습니다.');
   280→      }
   281→    } catch {
   282→      toast.error('저장 중 오류가 발생했습니다.');
   283→    } finally {
   284→      setIsSaving(false);
   285→    }
   286→  };
   287→
   288→  // 기본값으로 초기화
   289→  const handleReset = async () => {
   290→    setIsResetting(true);
   291→    try {
   292→      const res = await authenticatedFetch('/api/admin/mock/reset', {
   293→        method: 'POST',
   294→      });

   296→        setMockData({
   297→          chat: DEFAULT_CHAT,
   298→          image: DEFAULT_IMAGE,
   299→          future: DEFAULT_FUTURE,
