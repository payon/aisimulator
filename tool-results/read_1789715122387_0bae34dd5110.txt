   165→   165→// === 툴팁 위치 계산 ===
   166→   166→interface TooltipPosition {
   167→   167→  top: number;
   168→   168→  left: number;
   169→   169→  arrowPosition: 'top' | 'bottom' | 'left' | 'right';
   170→   170→}
   171→   171→
   172→   172→function calculateTooltipPosition(
   173→   173→  targetRect: DOMRect,
   174→   174→  position: GuideStep['position'],
   175→   175→  tooltipWidth: number,
   176→   176→  tooltipHeight: number
   177→   177→): TooltipPosition {
   178→   178→  const gap = 16;
   179→   179→  const vw = window.innerWidth;
   180→   180→  const vh = window.innerHeight;
   181→   181→  const padding = 12;
   182→   182→
   183→   183→  let top = 0;
   184→   184→  let left = 0;
   185→   185→  let arrowPosition = position;
   186→   186→
   187→   187→  switch (position) {
   188→   188→    case 'bottom':
   189→   189→      top = targetRect.bottom + gap;
   190→   190→      left = targetRect.left + targetRect.width / 2 - tooltipWidth / 2;
   191→   191→      break;
   192→   192→    case 'top':
   193→   193→      top = targetRect.top - tooltipHeight - gap;
   194→   194→      left = targetRect.left + targetRect.width / 2 - tooltipWidth / 2;
   195→   195→      break;
   196→   196→    case 'right':
   197→   197→      top = targetRect.top + targetRect.height / 2 - tooltipHeight / 2;
   198→   198→      left = targetRect.right + gap;
   199→   199→      break;
   200→   200→    case 'left':
   201→   201→      top = targetRect.top + targetRect.height / 2 - tooltipHeight / 2;
   202→   202→      left = targetRect.left - tooltipWidth - gap;
   203→   203→      break;
   204→   204→  }
   205→   205→
   206→   206→  // Clamp to viewport
   207→   207→  left = Math.max(padding, Math.min(left, vw - tooltipWidth - padding));
   208→   208→  top = Math.max(padding, Math.min(top, vh - tooltipHeight - padding));
   209→   209→
   210→   210→  // Fallback: if tooltip would be off-screen, reposition
   211→   211→  if (position === 'bottom' && top + tooltipHeight > vh - padding) {
   212→   212→    top = targetRect.top - tooltipHeight - gap;
   213→   213→    arrowPosition = 'top';
   214→   214→  }
   215→   215→  if (position === 'top' && top < padding) {
   216→   216→    top = targetRect.bottom + gap;
   217→   217→    arrowPosition = 'bottom';
   218→   218→  }
   219→   219→  if (position === 'right' && left + tooltipWidth > vw - padding) {
   220→   220→    left = targetRect.left - tooltipWidth - gap;
   221→   221→    arrowPosition = 'left';
   222→   222→  }
   223→   223→  if (position === 'left' && left < padding) {
   224→   224→    left = targetRect.right + tooltipWidth + gap;
   225→   225→    arrowPosition = 'right';
   226→   226→  }
   227→   227→
   228→   228→  return { top, left, arrowPosition };
   229→   229→}

   232→   232→function ArrowIcon({ direction, className }: { direction: GuideStep['position']; className?: string }) {
   233→   233→  switch (direction) {
   234→   234→    case 'top':

   236→   236→    case 'bottom':
   237→   237→      return <ChevronDown className={className} />;
   238→   238→    case 'left':
   239→   239→      return <ChevronLeft className={className} />;
   240→   240→    case 'right':
   241→   241→      return <ChevronRight className={className} />;
   242→   242→  }

   246→   246→export default function SeniorGuideTour() {
   247→   247→  const { active, currentStep, completed, start, stop, nextStep, prevStep, complete } = useGuideTourStore();
   248→   248→  const { voiceEnabled, readingSpeed } = useSettingsStore();
   249→   249→  const { speak, stop: stopTTS } = useSpeechSynthesis();
   250→   250→
   251→   251→  // 사이드바 제어 (모바일에서 자동 열기/닫기)
   252→   252→  const { isMobile, openMobile, setOpenMobile } = useSidebar();
   253→   253→
   254→   254→  const [targetRect, setTargetRect] = useState<DOMRect | null>(null);
   255→   255→  const [tooltipPos, setTooltipPos] = useState<TooltipPosition>({ top: 0, left: 0, arrowPosition: 'bottom' });
   256→   256→  const [showWelcome, setShowWelcome] = useState(false);
   257→   257→  const [showComplete, setShowComplete] = useState(false);
   258→   258→  const [stepNotFound, setStepNotFound] = useState(false);
   259→   259→  const [tooltipSize, setTooltipSize] = useState({ width: 320, height: 200 });
   260→   260→
   261→   261→  // 투어 중 사이드바를 자동으로 열었는지 추적 → 완료/스킵 시 원래대로 복원
   262→   262→  const sidebarAutoOpenedRef = useRef(false);
   263→   263→
   264→   264→  const tooltipRef = useRef<HTMLDivElement>(null);
   265→   265→  const resizeObserverRef = useRef<ResizeObserver | null>(null);
   266→   266→  const rafRef = useRef<number | null>(null);
   267→   267→  const retryTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   268→   268→  // 자기 자신을 재시도할 때 사용하는 ref (circular dependency 해결)
   269→   269→  const syncGuidePositionRef = useRef<() => void>(() => {});
   270→   270→
   271→   271→  // 현재 디바이스에 맞는 스텝 배열
   272→   272→  const steps = isMobile ? MOBILE_GUIDE_STEPS : DESKTOP_GUIDE_STEPS;
   273→   273→
   274→   274→  // Start tour -> show welcome screen
   275→   275→  const handleStartTour = useCallback(() => {
   276→   276→    setShowWelcome(true);
   277→   277→  }, []);
   278→   278→
   279→   279→  // Welcome -> begin actual tour
   280→   280→  const handleBeginTour = useCallback(() => {
   281→   281→    setShowWelcome(false);
   282→   282→    start();
   283→   283→  }, [start]);
   284→   284→
   285→   285→  // Skip from welcome
   286→   286→  const handleSkipWelcome = useCallback(() => {
   287→   287→    setShowWelcome(false);
   288→   288→    complete();
   289→   289→  }, [complete]);
   290→   290→
   291→   291→  // 모바일 사이드바 자동 열기 (필요한 스텝에서 호출)
   292→   292→  const ensureSidebarOpen = useCallback(() => {
   293→   293→    if (isMobile && !openMobile) {
   294→   294→      setOpenMobile(true);
   295→   295→      sidebarAutoOpenedRef.current = true;
   296→   296→    }
   297→   297→  }, [isMobile, openMobile, setOpenMobile]);
   298→   298→
   299→   299→  // 모바일 사이드바 자동 닫기 (투어 종료 시)
   300→   300→  const closeAutoOpenedSidebar = useCallback(() => {
   301→   301→    if (sidebarAutoOpenedRef.current && isMobile && openMobile) {
   302→   302→      setOpenMobile(false);
   303→   303→      sidebarAutoOpenedRef.current = false;
   304→   304→    }
   305→   305→  }, [isMobile, openMobile, setOpenMobile]);
   306→   306→
   307→   307→  // Combined: find target, update rect, and calculate tooltip position
   308→   308→  // Uses requestAnimationFrame to avoid synchronous setState in effects
   309→   309→  const syncGuidePosition = useCallback(() => {
   310→   310→    if (!active || showWelcome || showComplete) return;
   311→   311→
   312→   312→    const step = steps[currentStep];
   313→   313→    if (!step) {
   314→   314→      setShowComplete(true);
   315→   315→      return;
   316→   316→    }
   317→   317→
   318→   318→    // 모바일에서 사이드바가 필요한 스텝인데 사이드바가 닫혀 있으면 열고 재시도
   319→   319→    if (step.requiresSidebarOpen && isMobile && !openMobile) {
   320→   320→      ensureSidebarOpen();
   321→   321→      // 사이드바 Sheet 애니메이션 완료 후 재시도 (350ms 대기)
   322→   322→      if (retryTimerRef.current) clearTimeout(retryTimerRef.current);
   323→   323→      retryTimerRef.current = setTimeout(() => {
   324→   324→        syncGuidePositionRef.current();
   325→   325→      }, 350);
   326→   326→      return;
   327→   327→    }
   328→   328→
   329→   329→    const el = document.querySelector(step.targetSelector) as HTMLElement | null;
   330→   330→    if (!el) {
   331→   331→      // 모바일에서 사이드바 내부 요소를 못 찾으면 사이드바를 열고 재시도
   332→   332→      if (step.requiresSidebarOpen && isMobile) {
   333→   333→        ensureSidebarOpen();
   334→   334→        if (retryTimerRef.current) clearTimeout(retryTimerRef.current);
   335→   335→        retryTimerRef.current = setTimeout(() => {
   336→   336→          syncGuidePositionRef.current();
   337→   337→        }, 350);
   338→   338→        return;
   339→   339→      }
   340→   340→
   341→   341→      setStepNotFound(true);
   342→   342→      if (currentStep < steps.length - 1) {
   343→   343→        nextStep();
   344→   344→      } else {
   345→   345→        setShowComplete(true);
   346→   346→      }
   347→   347→      return;
   348→   348→    }
   349→   349→
   350→   350→    setStepNotFound(false);
   351→   351→    const rect = el.getBoundingClientRect();
   352→   352→    setTargetRect(rect);
   353→   353→
   354→   354→    // Scroll element into view if needed
   355→   355→    const isInViewport =
   356→   356→      rect.top >= 0 &&
   357→   357→      rect.left >= 0 &&
   358→   358→      rect.bottom <= window.innerHeight &&
   359→   359→      rect.right <= window.innerWidth;
   360→   360→
   361→   361→    if (!isInViewport) {
   362→   362→      el.scrollIntoView({ behavior: 'smooth', block: 'center', inline: 'center' });
   363→   363→    }
   364→   364→
   365→   365→    // Calculate tooltip position after getting rect
   366→   366→    // Use rAF for the scroll to settle
   367→   367→    rafRef.current = requestAnimationFrame(() => {
   368→   368→      const finalRect = el.getBoundingClientRect();
   369→   369→      setTargetRect(finalRect);
   370→   370→
   371→   371→      const tw = tooltipSize.width;
   372→   372→      const th = tooltipSize.height;
   373→   373→      const pos = calculateTooltipPosition(finalRect, step.position, tw, th);
   374→   374→      setTooltipPos(pos);
   375→   375→    });
   376→   376→  }, [active, currentStep, showWelcome, showComplete, nextStep, tooltipSize, steps, isMobile, openMobile, ensureSidebarOpen]);
   377→   377→
   378→   378→  // ref 업데이트 (setTimeout 재시도에서 사용)
   379→   379→  useEffect(() => {
   380→   380→    syncGuidePositionRef.current = syncGuidePosition;
   381→   381→  }, [syncGuidePosition]);
   382→   382→
   383→   383→  // Measure tooltip size when step changes
   384→   384→  useEffect(() => {
   385→   385→    if (tooltipRef.current) {
   386→   386→      const { width, height } = tooltipRef.current.getBoundingClientRect();
   387→   387→      setTooltipSize({ width, height });
   388→   388→    }
   389→   389→  }, [active, currentStep, showWelcome, showComplete]);
   390→   390→
   391→   391→  // Sync position on step change using rAF to avoid lint warning
   392→   392→  useEffect(() => {
   393→   393→    rafRef.current = requestAnimationFrame(() => {
   394→   394→      syncGuidePosition();
   395→   395→    });
   396→   396→    return () => {
   397→   397→      if (rafRef.current !== null) {
   398→   398→        cancelAnimationFrame(rafRef.current);
   399→   399→      }
   400→   400→      if (retryTimerRef.current !== null) {
   401→   401→        clearTimeout(retryTimerRef.current);
   402→   402→      }
   403→   403→    };
   404→   404→  }, [syncGuidePosition]);
   405→   405→
   406→   406→  // Resize observer + window resize/scroll listener for target element
   407→   407→  useEffect(() => {
   408→   408→    if (!active || showWelcome || showComplete) return;
   409→   409→
